# CREATE A MOVIE DATABASE IN POSTGRES USING
CREATE DATABASE movie;

# THEN CREATE A TABLE INSIDE movie DB
Table: public.movie_users
DROP TABLE IF EXISTS public.movie_users;
CREATE TABLE IF NOT EXISTS public.movie_users
(
    name character varying(15) COLLATE pg_catalog."default",
    email character varying(100) COLLATE pg_catalog."default" NOT NULL,
    password character varying(255) COLLATE pg_catalog."default",
    is_admin boolean DEFAULT false,
    actor text COLLATE pg_catalog."default",
    director text COLLATE pg_catalog."default",
    created timestamp with time zone DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT movie_users_pkey PRIMARY KEY (email)
)

TABLESPACE pg_default;

ALTER TABLE IF EXISTS public.movie_users
    OWNER to postgres;
